#define OPEN_GL_VERSION_MAJOR 3
#define OPEN_GL_VERSION_MINOR 3

typedef void (Gl_GenBuffersFn)(GLsizei buf_count, GLuint *bufs);
typedef void (Gl_BindBufferFn)(GLenum target, GLuint buf);
typedef void (Gl_BufferDataFn)(GLenum target, GLsizeiptr size,
                               const void *data, GLenum usage);
typedef U32 (Gl_CreateShaderFn)(GLenum shader_type);
typedef void (Gl_ShaderSourceFn)(GLuint shader, GLsizei string_count,
                                 const GLchar **string, const GLint *length);
typedef void (Gl_CompileShaderFn)(GLuint shader);
typedef GLuint (Gl_CreateProgramFn)(void);
typedef void (Gl_AttachShaderFn)(GLuint program, GLuint shader);
typedef void (Gl_LinkProgramFn)(GLuint program);
typedef void (Gl_UseProgramFn)(GLuint program);
typedef void (Gl_DeleteShaderFn)(GLuint shader);

typedef void (Gl_GetShaderivFn)(GLuint shader, GLenum param_name, GLint *params);
typedef void (Gl_GetShaderInfoLogFn)(GLuint shader, GLsizei max_length,
                                     GLsizei *length, GLchar *info_log);
typedef void (Gl_GetProgramivFn)(GLuint program, GLenum param_name, GLint *params);
typedef void (Gl_GetProgramInfoLogFn)(GLuint program, GLsizei max_length,
                                     GLsizei *length, GLchar *info_log);

typedef struct {
    // Mandatory functions for current compat
    Gl_GenBuffersFn *GenBuffers;
    Gl_BindBufferFn *BindBuffer;
    Gl_BufferDataFn *BufferData;
    Gl_CreateShaderFn *CreateShader;
    Gl_AttachShaderFn *AttachShader;
    Gl_DeleteShaderFn *DeleteShader;
    Gl_ShaderSourceFn *ShaderSource;
    Gl_CompileShaderFn *CompileShader;
    Gl_CreateProgramFn *CreateProgram;
    Gl_LinkProgramFn *LinkProgram;
    Gl_UseProgramFn *UseProgram;

    // Optional functions
    Gl_GetShaderivFn *GetShaderInputValue;
    Gl_GetShaderInfoLogFn *GetShaderInfoLog;
    Gl_GetProgramivFn *GetProgramInputValue;
    Gl_GetProgramInfoLogFn *GetProgramInfoLog;
} Gl_Funcs;

// NOTE: probs reusable across platforms -> nice abstraction might be possible
internal void glfw_bind_gl_funcs(Gl_Funcs *gl) {
    gl->GenBuffers = (Gl_GenBuffersFn *)glfwGetProcAddress("glGenBuffers");
    gl->BindBuffer = (Gl_BindBufferFn *)glfwGetProcAddress("glBindBuffer");
    gl->BufferData = (Gl_BufferDataFn *)glfwGetProcAddress("glBufferData");
    gl->CreateShader = (Gl_CreateShaderFn *)glfwGetProcAddress("glCreateShader");
    gl->ShaderSource = (Gl_ShaderSourceFn *)glfwGetProcAddress("glShaderSource");
    gl->CompileShader = (Gl_CompileShaderFn *)glfwGetProcAddress("glCompileShader");
    gl->CreateProgram = (Gl_CreateProgramFn *)glfwGetProcAddress("glCreateProgram");
    gl->AttachShader = (Gl_AttachShaderFn *)glfwGetProcAddress("glAttachShader");
    gl->LinkProgram = (Gl_LinkProgramFn *)glfwGetProcAddress("glLinkProgram");
    gl->UseProgram = (Gl_UseProgramFn *)glfwGetProcAddress("glUseProgram");
    gl->DeleteShader = (Gl_DeleteShaderFn *)glfwGetProcAddress("glDeleteShader");

    gl->GetShaderInputValue = (Gl_GetShaderivFn *)glfwGetProcAddress("glGetShaderiv");
    gl->GetShaderInfoLog = (Gl_GetShaderInfoLogFn *)glfwGetProcAddress("glGetShaderInfoLog");
    gl->GetProgramInputValue = (Gl_GetProgramivFn *)glfwGetProcAddress("glGetProgramiv");
    gl->GetProgramInfoLog = (Gl_GetProgramInfoLogFn *)glfwGetProcAddress("glGetProgramInfoLog");

    B32 necessary_func_missing = !gl->GenBuffers || !gl->BindBuffer || !gl->BufferData
        || !gl->CreateShader || !gl->DeleteShader
        || !gl->ShaderSource || !gl->CompileShader || !gl->AttachShader
        || !gl->CreateProgram || !gl->LinkProgram || !gl->UseProgram;
    if (necessary_func_missing) {
        // TODO: logging + exit?
        // TODO: compatibility -> e.g. use older gl version with fixed func pipeline or sth??
    }

    const char *shared_shader_header = "#version 330 core\n";
    const char *vertex_shader_src_no_header =
        "layout (location = 0) in vec3 aPos;\n"
        "void main() {\n"
        "   gl_Position = vec4(aPos.x, aPos.y, aPos.z, 1.0);\n"
        "}\0";
    const char *vertex_shader_src[]
        = {shared_shader_header, vertex_shader_src_no_header};
    U32 vertex_shader = gl->CreateShader(GL_VERTEX_SHADER);
    gl->ShaderSource(vertex_shader, 2, vertex_shader_src, 0);
    gl->CompileShader(vertex_shader);

#if EDITORE_DEBUG
    char info_log[512];
    if (gl->GetShaderInputValue && gl->GetShaderInfoLog) {
        int compiled;
        gl->GetShaderInputValue(vertex_shader, GL_COMPILE_STATUS, &compiled);
        if (!compiled) {
            gl->GetShaderInfoLog(vertex_shader, ArrayCount(info_log), 0, info_log);
            fprintf(stderr, "Vertex shader compile failed: %s\n", info_log);
        }
    } else {
        fprintf(stderr, "Missing functions to check failing compile of shader\n");
    }
#endif // EDITORE_DEBUG

    const char *fragment_shader_src_no_header =
        "out vec4 FragColor;\n"
        "void main() {\n"
        "   FragColor = vec4(1.0f, 0.5f, 0.2f, 1.0f);\n"
        "}\0";
    const char *fragment_shader_src[]
        = {shared_shader_header, fragment_shader_src_no_header};
    U32 fragment_shader = gl->CreateShader(GL_FRAGMENT_SHADER);
    gl->ShaderSource(fragment_shader, 2, fragment_shader_src, 0);
    gl->CompileShader(fragment_shader);

#if EDITORE_DEBUG
    if (gl->GetShaderInputValue && gl->GetShaderInfoLog) {
        int compiled;
        gl->GetShaderInputValue(vertex_shader, GL_COMPILE_STATUS, &compiled);
        if (!compiled) {
            gl->GetShaderInfoLog(vertex_shader, ArrayCount(info_log), 0, info_log);
            fprintf(stderr, "Fragment shader compile failed: %s\n", info_log);
        }
    } else {
        fprintf(stderr, "Missing functions to check failing compile of shader\n");
    }
#endif // EDITORE_DEBUG

    U32 shader_program = gl->CreateProgram();
    gl->AttachShader(shader_program, vertex_shader);
    gl->AttachShader(shader_program, fragment_shader);
    gl->LinkProgram(shader_program);

#if EDITORE_DEBUG
    if (gl->GetProgramInputValue && gl->GetProgramInfoLog) {
        int linked;
        gl->GetProgramInputValue(shader_program, GL_LINK_STATUS, &linked);
        if(!linked) {
            gl->GetProgramInfoLog(shader_program, ArrayCount(info_log), 0, info_log);
            fprintf(stderr, "Shader pgm linking failed: %s\n", info_log);
        }
    } else {
        fprintf(stderr, "Missing functions to check failing linking of shader pgm\n");
    }
#endif // EDITORE_DEBUG

    gl->UseProgram(shader_program);
    gl->DeleteShader(vertex_shader);
    gl->DeleteShader(fragment_shader);
}

internal void glfw_framebuffer_size_cb(GLFWwindow* window, int width, int height) {
    glViewport(0, 0, width, height);
}

internal GLFWwindow *glfw_init(void) {
    glfwInit();
    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, OPEN_GL_VERSION_MAJOR);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, OPEN_GL_VERSION_MINOR);
    glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);

    S32 window_width = 1200;
    S32 window_height = 780;
    GLFWwindow *window = glfwCreateWindow(window_width, window_height, "AuzioEditore", 0, 0);
    if (window == 0) {
        // TODO: logging
        fprintf(stderr, "Failed to create GLFW window\n");
        return window;
    }
    glfwMakeContextCurrent(window);

    glViewport(0, 0, window_width, window_height);
    glfwSetFramebufferSizeCallback(window, glfw_framebuffer_size_cb);

    Gl_Funcs gl = {0};
    glfw_bind_gl_funcs(&gl);

    float vertices[] = {
        -0.5f, -0.5f, 0.0f,
         0.5f, -0.5f, 0.0f,
         0.0f,  0.5f, 0.0f
    };
    U32 vbo;
    gl.GenBuffers(1, &vbo);
    gl.BindBuffer(GL_ARRAY_BUFFER, vbo);
    gl.BufferData(GL_ARRAY_BUFFER, sizeof(vertices), vertices, GL_STATIC_DRAW);

    return window;
}

int main(int argc, char **argv) {
    GLFWwindow *window = glfw_init();

    while(!glfwWindowShouldClose(window)) {
        // TODO: event based scan_key_codes + char pressed -> probs better via some native window handle?
        if(
            glfwGetKey(window, GLFW_KEY_ESCAPE) == GLFW_PRESS
        ) {
            glfwSetWindowShouldClose(window, true);
        }

        glClearColor(0.2f, 0.3f, 0.3f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT);

        glfwPollEvents();
        glfwSwapBuffers(window);
    }

    return 0;
}
